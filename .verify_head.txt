30| /* Per-browser resume storage. localStorage is naturally isolated per
31|    browser/device (no accounts needed), and the JSON wrapper leaves room
32|    for multiple templates later. All access is guarded so private-mode
33|    browsers or corrupted data never crash the app. */

35| const STORAGE_KEY = "portfolio_resume_data";
36| const RESUME_ID = "resume"; // current template slot; more can be added later
37| 
38| function readSavedResume(): string | null {
39|   try {
40|     const raw = window.localStorage.getItem(STORAGE_KEY);
41|     if (!raw) return null;
42|     const parsed = JSON.parse(raw) as { v?: number; resumes?: Record<string, string> };
43|     const html = parsed?.resumes?.[RESUME_ID];
44|     return typeof html === "string" && html.trim() ? html : null;
45|   } catch {
46|     return null; // malformed data or unavailable storage → use default
47|   }
48| }
49| 
50| function writeSavedResume(html: string) {
51|   try {
52|     let data: { v: number; resumes: Record<string, string> } = { v: 1, resumes: {} };
53|     try {
54|       const raw = window.localStorage.getItem(STORAGE_KEY);
55|       if (raw) {
56|         const parsed = JSON.parse(raw);
57|         if (parsed && typeof parsed === "object") {
58|           data = { v: 1, resumes: { ...(parsed.resumes ?? {}) } };
59|         }
60|       }
61|     } catch {
62|       /* keep fresh wrapper if old data was corrupted */
63|     }
64|     data.resumes[RESUME_ID] = html;
65|     window.localStorage.setItem(STORAGE_KEY, JSON.stringify(data));
66|     return true;
67|   } catch {
68|     return false; // storage unavailable (e.g. some private modes)
69|   }
70| }
71| 
72| function removeSavedResume() {
73|   try {
74|     window.localStorage.removeItem(STORAGE_KEY);
75|   } catch {
76|     /* storage unavailable — nothing to remove */
77|   }
78| }

140| /* Closest block-level ancestor inside the paper (or the paper itself). */
141| function getBlock(root: HTMLElement, node: Node | null): HTMLElement | null {
142|   let cur: Node | null = node;
143|   if (cur && cur.nodeType === Node.TEXT_NODE) cur = cur.parentNode;
144|   while (cur && cur !== root) {
145|     if (cur.parentNode === root && cur instanceof HTMLElement) return cur;
146|     cur = cur.parentNode;
147|   }
148|   return root;
149| }

151| /* Top-level blocks of the paper intersected by a range. For a collapsed
152|    caret this yields the single block containing the caret. */
153| function blocksInRange(root: HTMLElement, range: Range): HTMLElement[] {
154|   return (Array.from(root.children) as HTMLElement[]).filter((b) =>
155|     range.intersectsNode(b)
156|   );
157| }

159| /* Paragraph-level formatting driven EXPLICITLY by the given range —
160|    never re-reads the (possibly cleared) window selection. */
161| function applyBlocksStyle(root: HTMLElement, range: Range, prop: string, value: string) {
162|   const targets = blocksInRange(root, range);
163|   if (targets.length > 0) {
164|     targets.forEach((b) => b.style.setProperty(prop, value));
165|     return;
166|   }
167|   const fallback = getBlock(root, range.startContainer);
168|   if (fallback) (fallback as HTMLElement).style.setProperty(prop, value);
169| }

171| /* Wrap exactly the text selected by `range` in a span with the given
172|    styles — so formatting applies ONLY to the highlighted text.
173| 
174|    Uses a fragment-extract-wrap-reinsert approach that works for selections
175|    spanning any number of nodes/blocks. Falls back to manual DOM surgery
176|    when surroundContents is not supported for the range shape. */
177| function wrapRangeInline(root: HTMLElement, range: Range, styles: Record<string, string>) {
178|   if (range.collapsed || !root.contains(range.commonAncestorContainer)) return;
179| 
180|   const didApply = trySurround(range, styles);
181|   if (didApply) return;
182| 
183|   // Robust fallback for ranges that surroundContents cannot handle
184|   // (e.g. selections spanning multiple nodes/elements).
185|   applyViaExtract(range, styles);
186| }
187| 
188| function trySurround(range: Range, styles: Record<string, string>) {
189|   // Normalize collapsed whitespace-only ranges.
190|   if (range.collapsed) return false;
191|   try {
192|     const span = document.createElement("span");
193|     for (const [prop, value] of Object.entries(styles)) {
194|       span.style.setProperty(prop, value);
195|     }
196|     range.surroundContents(span);
197|     return true;
198|   } catch {
199|     return false;
200|   }
201| }
202| 
203| function applyViaExtract(range: Range, styles: Record<string, string>) {
204|   const startContainer = range.startContainer;
205|   const endContainer = range.endContainer;
206|   if (!startContainer || !endContainer) return;
207| 
208|   // Compute the selected fragment across all nodes the range touches.
209|   const frag = range.cloneContents();
210| 
211|   // Remove the original selected content from the document.
212|   range.deleteContents();
213| 
214|   // Build the styled wrapper.
215|   const span = document.createElement("span");
216|   for (const [prop, value] of Object.entries(styles)) {
217|     span.style.setProperty(prop, value);
218|   }
219|   span.appendChild(frag);
220| 
221|   // Re-insert the wrapper at the (now-collapsed) range position.
222|   const insertPoint = range.startContainer;
223|   const insertOffset = range.startOffset;
224|   if (insertPoint.nodeType === Node.TEXT_NODE) {
225|     const parent = insertPoint.parentNode;
226|     if (parent) {
227|       const textBefore = (insertPoint as Text).splitText(insertOffset);
228|       parent.insertBefore(span, textBefore);
229|     } else {
230|       // Defensive: insert at the range start as a fallback.
231|       const parentEl = insertPoint instanceof Element ? insertPoint : (insertPoint.parentNode as Node);
232|       if (parentEl) {
233|         const ref = insertPoint.nextSibling;
234|         parentEl.insertBefore(span, ref);
235|       }
236|     }
237|   } else if (insertPoint instanceof Element) {
238|     insertPoint.insertBefore(span, insertPoint.childNodes[insertOffset] as Node | null);
239|   } else if (insertPoint.parentNode) {
240|     insertPoint.parentNode.insertBefore(span, insertPoint.nextSibling);
241|   }
242| 
243|   // Restore selection to the newly inserted span's contents.
244|   try {
245|     const sel = window.getSelection();
246|     if (sel) {
247|       const r = document.createRange();
248|       if (span.firstChild) {
249|         r.selectNodeContents(span);
250|       } else {
251|         r.setStart(span, 0);
252|         r.collapse(true);
253|       }
254|       sel.removeAllRanges();
255|       sel.addRange(r);
256|     }
257|   } catch {
258|     /* selection restore is best-effort */
259|   }
260| }

270| /* Read the effective bold/italic/underline state of the first text
271|    node touched by the range (used for manual toggles). */
272| function inlineStateOfRange(range: Range): { bold: boolean; italic: boolean; underline: boolean } {
273|   let node: Node | null = range.startContainer;
274|   if (node.nodeType === Node.TEXT_NODE) {
275|     node = node.parentNode;
276|   }
277|   const el = (node instanceof HTMLElement ? node : node?.parentElement) ?? null;
278|   if (!el) return { bold: false, italic: false, underline: false };
279|   const cs = window.getComputedStyle(el);
280|   const weight = parseInt(cs.fontWeight, 10);
281|   return {
282|     bold: Number.isFinite(weight) ? weight >= 600 : cs.fontWeight === "bold",
283|     italic: cs.fontStyle === "italic",
284|     underline: (cs.textDecorationLine ?? cs.textDecoration).toString().includes("underline"),
285|   };
286| }

290| /* A section starts at an .rsec heading and ends
291|    right before the next .rsec heading (or the paper's end). */
292| function getSectionRange(root: HTMLElement, node: Node | null): { start: HTMLElement; end: HTMLElement | null } | null {
293|   const block = getBlock(root, node);
294|   if (!block || block === root) return null;
295|   let start: HTMLElement | null = null;
296|   let el = block.previousElementSibling as HTMLElement | null;
297|   while (el) {
298|     if (el.classList.contains("rsec")) {
299|       start = el;
300|       break;
301|     }
302|     el = el.previousElementSibling as HTMLElement | null;
303|   }
304|   if (!start) start = (root.firstElementChild as HTMLElement) ?? block;
305|   let end: HTMLElement | null = null;
306|   el = start.nextElementSibling as HTMLElement | null;
307|   while (el) {
308|     if (el.classList.contains("rsec")) {
309|       end = el;
310|       break;
311|     }
312|     el = el.nextElementSibling as HTMLElement | null;
313|   }
314|   return { start, end };
315| }
