468|   /* Insert a node at the given container/offset. For TEXT_NODEs we use
469|      splitText so the marker sits at the text offset without corrupting
470|      content. For ELEMENTs we insert before the child at the offset. */
471|   const insertNodeAtPosition = (
472|     paper: HTMLElement,
473|     node: Node,
474|     container: Node,
475|     offset: number
476|   ) => {
477|     if (!container || !paper) return;
478|     if (container.nodeType === Node.TEXT_NODE) {
479|       const textNode = container as Text;
480|       const parent = textNode.parentNode;
481|       if (!parent) return;
482|       const rest = textNode.splitText(offset);
483|       parent.insertBefore(node, rest);
484|     } else if (container instanceof Element) {
485|       const child = container.childNodes[offset] as Node | null;
486|       container.insertBefore(node, child);
487|     } else if (container.parentNode) {
488|       container.parentNode.insertBefore(node, container.nextSibling);
489|     }
490|   };
